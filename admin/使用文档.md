## 通用组件

### 1. Pagination 分页组件

**文件位置**: `src/components/Pagination.vue`

#### 基本用法

```vue
<script setup lang="ts">
import Pagination from '@/components/Pagination.vue'
import { ref } from 'vue'

const page = ref(1)
const pageSize = ref(10)
const total = ref(100)

const handlePageChange = (newPage: number, newPageSize: number) => {
  page.value = newPage
  pageSize.value = newPageSize
  // 重新加载数据
  loadData()
}
</script>

<template>
  <Pagination
    :total="total"
    :page="page"
    :page-size="pageSize"
    @change="handlePageChange"
  />
</template>
```

#### Props

| 参数 | 类型 | 默认值 | 说明 |
|------|------|--------|------|
| total | number | - | 总条数（必填） |
| page | number | - | 当前页码（必填） |
| pageSize | number | - | 每页条数（必填） |
| pageSizes | number[] | [10, 20, 50, 100] | 每页条数选项 |

#### Events

| 事件名 | 参数 | 说明 |
|--------|------|------|
| change | (page: number, pageSize: number) | 页码或每页条数改变时触发 |

### 3. BatchActions 批量操作组件

**文件位置**: `src/components/BatchActions.vue`

#### 基本用法

```vue
<script setup lang="ts">
import BatchActions from '@/components/BatchActions.vue'
import { ref } from 'vue'

const selectedIds = ref<number[]>([])

const handleBatchDelete = async () => {
  await http.delete('/api/items/batch', { data: { ids: selectedIds.value } })
  ElMessage.success('批量删除成功')
  loadData()
}

const handleBatchStatusChange = async (status: string) => {
  await http.patch('/api/items/batch', { ids: selectedIds.value, status })
  ElMessage.success('批量修改成功')
  loadData()
}
</script>

<template>
  <BatchActions
    :selected-ids="selectedIds"
    @delete="handleBatchDelete"
    @status-change="handleBatchStatusChange"
  >
    <template #extra="{ hasSelection, count }">
      <el-button :disabled="!hasSelection" @click="handleExport">
        导出选中 ({{ count }})
      </el-button>
    </template>
  </BatchActions>
</template>
```

#### Props

| 参数 | 类型 | 说明 |
|------|------|------|
| selectedIds | number[] | 选中的ID数组 |

#### Events

| 事件名 | 参数 | 说明 |
|--------|------|------|
| delete | - | 批量删除 |
| statusChange | status: string | 批量修改状态 |

#### Slots

| 插槽名 | 参数 | 说明 |
|--------|------|------|
| extra | { hasSelection, count } | 自定义额外操作按钮 |

## 组合式函数

### usePagination

**文件位置**: `src/composables/usePagination.ts`

#### 基本用法

```vue
<script setup lang="ts">
import { usePagination } from '@/composables/usePagination'
import http from '@/utils/http'

// 定义API请求函数
const fetchData = (params: any) => {
  return http.get('/api/news', { params })
}

// 使用 usePagination
const { data, total, loading, page, pageSize, fetch, handlePageChange } = usePagination(fetchData)

// 搜索功能
const searchForm = ref({ title: '', category: '' })
const handleSearch = () => {
  fetch(searchForm.value)
}

// 初始加载
fetch()
</script>

<template>
  <el-table :data="data" v-loading="loading">
    <!-- 表格列 -->
  </el-table>

  <Pagination
    :total="total"
    :page="page"
    :page-size="pageSize"
    @change="handlePageChange"
  />
</template>
```

#### 返回值

| 属性 | 类型 | 说明 |
|------|------|------|
| data | Ref<any[]> | 数据列表 |
| total | Ref<number> | 总条数 |
| loading | Ref<boolean> | 加载状态 |
| page | Ref<number> | 当前页码 |
| pageSize | Ref<number> | 每页条数 |
| fetch | (params?) => Promise<void> | 加载数据函数 |
| handlePageChange | (page, pageSize) => void | 分页改变处理函数 |

#### API 响应格式要求

```typescript
// 后端需返回以下格式
{
  code: 200,
  data: {
    items: [...],  // 数据列表
    total: 100     // 总条数
  }
}

// 或简化格式
{
  code: 200,
  data: [...]  // 直接返回数组，total 将使用数组长度
}
```

## 页面开发指南

### 标准 CRUD 页面模板

```vue
<script setup lang="ts">
import { ref } from 'vue'
import { ElMessage, ElMessageBox } from 'element-plus'
import { usePagination } from '@/composables/usePagination'
import { requiredRule } from '@/utils/validate'
import Pagination from '@/components/Pagination.vue'
import http from '@/utils/http'

// 搜索表单
const searchForm = ref({ keyword: '' })

// 对话框
const dialogVisible = ref(false)
const dialogTitle = ref('新增')
const formRef = ref()
const formData = ref({ name: '', description: '' })

// 验证规则
const rules = {
  name: [requiredRule('名称')]
}

// 分页数据
const { data, total, loading, page, pageSize, fetch, handlePageChange } = usePagination(
  (params) => http.get('/api/items', { params })
)

// 搜索
const handleSearch = () => fetch(searchForm.value)
const handleReset = () => {
  searchForm.value = { keyword: '' }
  fetch()
}

// 新增
const handleAdd = () => {
  dialogTitle.value = '新增'
  formData.value = { name: '', description: '' }
  dialogVisible.value = true
}

// 编辑
const handleEdit = (row: any) => {
  dialogTitle.value = '编辑'
  formData.value = { ...row }
  dialogVisible.value = true
}

// 删除
const handleDelete = (row: any) => {
  ElMessageBox.confirm('确定删除吗？', '提示', { type: 'warning' }).then(async () => {
    await http.delete(`/api/items/${row.id}`)
    ElMessage.success('删除成功')
    fetch()
  })
}

// 保存
const handleSave = async () => {
  await formRef.value.validate()
  if (formData.value.id) {
    await http.patch(`/api/items/${formData.value.id}`, formData.value)
  } else {
    await http.post('/api/items', formData.value)
  }
  ElMessage.success('保存成功')
  dialogVisible.value = false
  fetch()
}

// 初始加载
fetch()
</script>

<template>
  <div class="page-container">
    <!-- 搜索区域 -->
    <el-card shadow="never" class="mb-md">
      <el-form :inline="true" :model="searchForm">
        <el-form-item label="关键词">
          <el-input v-model="searchForm.keyword" placeholder="请输入关键词" clearable />
        </el-form-item>
        <el-form-item>
          <el-button type="primary" @click="handleSearch">查询</el-button>
          <el-button @click="handleReset">重置</el-button>
        </el-form-item>
      </el-form>
    </el-card>

    <!-- 列表区域 -->
    <el-card shadow="never">
      <div class="flex-between mb-md">
        <h3>数据列表</h3>
        <el-button type="primary" @click="handleAdd">新增</el-button>
      </div>

      <el-table :data="data" v-loading="loading" stripe>
        <el-table-column prop="name" label="名称" />
        <el-table-column prop="description" label="描述" />
        <el-table-column label="操作" width="150" fixed="right">
          <template #default="{ row }">
            <el-button text type="primary" size="small" @click="handleEdit(row)">编辑</el-button>
            <el-button text type="danger" size="small" @click="handleDelete(row)">删除</el-button>
          </template>
        </el-table-column>
      </el-table>

      <Pagination :total="total" :page="page" :page-size="pageSize" @change="handlePageChange" />
    </el-card>

    <!-- 编辑对话框 -->
    <el-dialog v-model="dialogVisible" :title="dialogTitle" width="600px">
      <el-form ref="formRef" :model="formData" :rules="rules" label-width="100px">
        <el-form-item label="名称" prop="name">
          <el-input v-model="formData.name" placeholder="请输入名称" />
        </el-form-item>
        <el-form-item label="描述">
          <el-input v-model="formData.description" type="textarea" :rows="3" />
        </el-form-item>
      </el-form>
      <template #footer>
        <el-button @click="dialogVisible = false">取消</el-button>
        <el-button type="primary" @click="handleSave">保存</el-button>
      </template>
    </el-dialog>
  </div>
</template>

<style scoped lang="scss">
.page-container {
  h3 {
    margin: 0;
    font-size: 16px;
    font-weight: 500;
  }
}
</style>
```

## 常见问题

### Q1: 分页数据不显示？

**检查后端响应格式**：
```json
{
  "code": 200,
  "data": {
    "items": [...],
    "total": 100
  }
}
```

### Q2: 图片上传失败？

**检查上传接口**：
- 接口地址：`/api/upload`
- 请求方法：`POST`
- 请求格式：`multipart/form-data`
- 响应格式：`{ code: 200, data: { url: '图片URL' } }`

### Q3: 表单验证不生效？

**检查以下几点**：
1. `el-form` 是否绑定了 `ref`
2. `el-form-item` 是否设置了 `prop` 属性
3. 是否调用了 `formRef.value.validate()`

### Q4: 样式变量不生效？

**确保在 `<style>` 中使用 `lang="scss"`**：
```vue
<style scoped lang="scss">
.container {
  padding: $spacing-md;  /* 需要 lang="scss" */
}
</style>
```
